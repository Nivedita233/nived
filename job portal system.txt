import java.util.ArrayList;
import java.util.Scanner;

// ---------------- JOB CLASS ----------------
class Job {
    private int jobId;
    private String company;
    private String title;
    private String location;
    private String salary;
    private String qualification;

    public Job(int jobId, String company, String title,
               String location, String salary, String qualification) {
        this.jobId = jobId;
        this.company = company;
        this.title = title;
        this.location = location;
        this.salary = salary;
        this.qualification = qualification;
    }

    public int getJobId() {
        return jobId;
    }

    public String getCompany() {
        return company;
    }

    public String getTitle() {
        return title;
    }

    public void displayJob() {
        System.out.println("--------------------------------------------");
        System.out.println("Job ID        : " + jobId);
        System.out.println("Company       : " + company);
        System.out.println("Job Title     : " + title);
        System.out.println("Location      : " + location);
        System.out.println("Salary        : " + salary);
        System.out.println("Qualification : " + qualification);
    }
}

// ---------------- USER CLASS ----------------
class User {
    private String name;
    private String email;
    private String password;
    private String qualification;

    public User(String name, String email,
                String password, String qualification) {
        this.name = name;
        this.email = email;
        this.password = password;
        this.qualification = qualification;
    }

    public String getEmail() {
        return email;
    }

    public String getPassword() {
        return password;
    }

    public String getName() {
        return name;
    }

    public String getQualification() {
        return qualification;
    }
}

// ---------------- APPLICATION CLASS ----------------
class Application {
    private String applicantName;
    private String email;
    private String qualification;
    private int jobId;
    private String company;
    private String jobTitle;
    private String status;

    public Application(String applicantName, String email,
                       String qualification, int jobId,
                       String company, String jobTitle) {

        this.applicantName = applicantName;
        this.email = email;
        this.qualification = qualification;
        this.jobId = jobId;
        this.company = company;
        this.jobTitle = jobTitle;
        this.status = "Applied";
    }

    public void displayApplication() {
        System.out.println("--------------------------------------------");
        System.out.println("Applicant     : " + applicantName);
        System.out.println("Email         : " + email);
        System.out.println("Qualification : " + qualification);
        System.out.println("Job ID        : " + jobId);
        System.out.println("Company       : " + company);
        System.out.println("Job Title     : " + jobTitle);
        System.out.println("Status        : " + status);
    }
}

// ---------------- MAIN CLASS ----------------
public class JobPortalSystem {

    static Scanner sc = new Scanner(System.in);

    static ArrayList<Job> jobs = new ArrayList<>();
    static ArrayList<User> users = new ArrayList<>();
    static ArrayList<Application> applications = new ArrayList<>();

    // ---------------- ADD SAMPLE JOBS ----------------
    static void addSampleJobs() {

        jobs.add(new Job(
                101,
                "TCS",
                "Java Developer",
                "Bangalore",
                "₹5 LPA",
                "B.E/B.Tech"
        ));

        jobs.add(new Job(
                102,
                "Infosys",
                "Software Engineer",
                "Pune",
                "₹6 LPA",
                "B.E/B.Tech/MCA"
        ));

        jobs.add(new Job(
                103,
                "Wipro",
                "Web Developer",
                "Hyderabad",
                "₹4.5 LPA",
                "BCA/B.Tech"
        ));

        jobs.add(new Job(
                104,
                "Accenture",
                "Data Analyst",
                "Mumbai",
                "₹7 LPA",
                "B.E/B.Tech/MCA"
        ));
    }

    // ---------------- REGISTER USER ----------------
    static void registerUser() {

        sc.nextLine();

        System.out.println("\n========== USER REGISTRATION ==========");

        System.out.print("Enter name: ");
        String name = sc.nextLine();

        System.out.print("Enter email: ");
        String email = sc.nextLine();

        // Check duplicate email
        for (User u : users) {
            if (u.getEmail().equalsIgnoreCase(email)) {
                System.out.println("Email already registered!");
                return;
            }
        }

        System.out.print("Create password: ");
        String password = sc.nextLine();

        System.out.print("Enter qualification: ");
        String qualification = sc.nextLine();

        users.add(new User(
                name,
                email,
                password,
                qualification
        ));

        System.out.println("\nRegistration successful!");
    }

    // ---------------- USER LOGIN ----------------
    static User userLogin() {

        sc.nextLine();

        System.out.println("\n========== USER LOGIN ==========");

        System.out.print("Enter email: ");
        String email = sc.nextLine();

        System.out.print("Enter password: ");
        String password = sc.nextLine();

        for (User u : users) {

            if (u.getEmail().equalsIgnoreCase(email)
                    && u.getPassword().equals(password)) {

                System.out.println("\nLogin successful!");
                System.out.println("Welcome, " + u.getName());

                return u;
            }
        }

        System.out.println("Invalid email or password.");
        return null;
    }

    // ---------------- VIEW JOBS ----------------
    static void viewJobs() {

        System.out.println("\n========== AVAILABLE JOBS ==========");

        if (jobs.isEmpty()) {
            System.out.println("No jobs available.");
            return;
        }

        for (Job job : jobs) {
            job.displayJob();
        }
    }

    // ---------------- SEARCH JOB ----------------
    static void searchJob() {

        sc.nextLine();

        System.out.println("\n========== SEARCH JOB ==========");

        System.out.print("Enter job title/company/location: ");
        String search = sc.nextLine().toLowerCase();

        boolean found = false;

        for (Job job : jobs) {

            if (job.getTitle().toLowerCase().contains(search)
                    || job.getCompany().toLowerCase().contains(search)) {

                job.displayJob();
                found = true;
            }
        }

        if (!found) {
            System.out.println("No matching jobs found.");
        }
    }

    // ---------------- APPLY JOB ----------------
    static void applyJob(User user) {

        viewJobs();

        System.out.print("\nEnter Job ID to apply: ");
        int id = sc.nextInt();

        for (Job job : jobs) {

            if (job.getJobId() == id) {

                applications.add(
                        new Application(
                                user.getName(),
                                user.getEmail(),
                                user.getQualification(),
                                job.getJobId(),
                                job.getCompany(),
                                job.getTitle()
                        )
                );

                System.out.println("\nApplication submitted successfully!");
                return;
            }
        }

        System.out.println("Invalid Job ID.");
    }

    // ---------------- MY APPLICATIONS ----------------
    static void viewMyApplications(User user) {

        boolean found = false;

        System.out.println("\n========== MY APPLICATIONS ==========");

        for (Application a : applications) {

            // Application information is displayed
            a.displayApplication();
            found = true;
        }

        if (!found) {
            System.out.println("No applications found.");
        }
    }

    // ---------------- ADD JOB BY ADMIN ----------------
    static void addJobByAdmin() {

        sc.nextLine();

        System.out.println("\n========== ADD NEW JOB ==========");

        System.out.print("Enter Job ID: ");
        int id = sc.nextInt();
        sc.nextLine();

        System.out.print("Enter Company Name: ");
        String company = sc.nextLine();

        System.out.print("Enter Job Title: ");
        String title = sc.nextLine();

        System.out.print("Enter Location: ");
        String location = sc.nextLine();

        System.out.print("Enter Salary: ");
        String salary = sc.nextLine();

        System.out.print("Enter Qualification: ");
        String qualification = sc.nextLine();

        jobs.add(
                new Job(
                        id,
                        company,
                        title,
                        location,
                        salary,
                        qualification
                )
        );

        System.out.println("\nJob added successfully!");
    }

    // ---------------- ADMIN LOGIN ----------------
    static boolean adminLogin() {

        sc.nextLine();

        System.out.println("\n========== ADMIN LOGIN ==========");

        System.out.print("Username: ");
        String username = sc.nextLine();

        System.out.print("Password: ");
        String password = sc.nextLine();

        if (username.equals("admin")
                && password.equals("admin123")) {

            System.out.println("\nAdmin login successful!");
            return true;
        }

        System.out.println("Invalid admin credentials.");
        return false;
    }

    // ---------------- ADMIN MENU ----------------
    static void adminMenu() {

        int choice;

        do {

            System.out.println("\n==================================");
            System.out.println("           ADMIN MENU");
            System.out.println("==================================");
            System.out.println("1. Add New Job");
            System.out.println("2. View All Jobs");
            System.out.println("3. View Applications");
            System.out.println("4. Logout");
            System.out.println("==================================");

            System.out.print("Enter choice: ");
            choice = sc.nextInt();

            switch (choice) {

                case 1:
                    addJobByAdmin();
                    break;

                case 2:
                    viewJobs();
                    break;

                case 3:

                    if (applications.isEmpty()) {
                        System.out.println("No applications received.");
                    } else {

                        System.out.println(
                                "\n========== ALL APPLICATIONS =========="
                        );

                        for (Application a : applications) {
                            a.displayApplication();
                        }
                    }

                    break;

                case 4:
                    System.out.println("Admin logged out.");
                    break;

                default:
                    System.out.println("Invalid choice.");
            }

        } while (choice != 4);
    }

    // ---------------- USER MENU ----------------
    static void userMenu(User user) {

        int choice;

        do {

            System.out.println("\n==================================");
            System.out.println("            USER MENU");
            System.out.println("==================================");
            System.out.println("1. View Available Jobs");
            System.out.println("2. Search Job");
            System.out.println("3. Apply for Job");
            System.out.println("4. My Applications");
            System.out.println("5. Logout");
            System.out.println("==================================");

            System.out.print("Enter choice: ");
            choice = sc.nextInt();

            switch (choice) {

                case 1:
                    viewJobs();
                    break;

                case 2:
                    searchJob();
                    break;

                case 3:
                    applyJob(user);
                    break;

                case 4:
                    viewMyApplications(user);
                    break;

                case 5:
                    System.out.println("Logged out successfully.");
                    break;

                default:
                    System.out.println("Invalid choice.");
            }

        } while (choice != 5);
    }

    // ---------------- MAIN METHOD ----------------
    public static void main(String[] args) {

        addSampleJobs();

        int choice;

        do {

            System.out.println("\n");
            System.out.println("==========================================");
            System.out.println("          ONLINE JOB PORTAL SYSTEM");
            System.out.println("==========================================");
            System.out.println("1. User Registration");
            System.out.println("2. User Login");
            System.out.println("3. Admin Login");
            System.out.println("4. Exit");
            System.out.println("==========================================");

            System.out.print("Enter your choice: ");
            choice = sc.nextInt();

            switch (choice) {

                case 1:
                    registerUser();
                    break;

                case 2:

                    User user = userLogin();

                    if (user != null) {
                        userMenu(user);
                    }

                    break;

                case 3:

                    if (adminLogin()) {
                        adminMenu();
                    }

                    break;

                case 4:
                    System.out.println(
                            "\nThank you for using Online Job Portal System!"
                    );
                    break;

                default:
                    System.out.println("Invalid choice. Try again.");
            }

        } while (choice != 4);

        sc.close();
    }
}
